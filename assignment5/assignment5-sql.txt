SELECT li.order_id, SUM(p.price * li.quantity) AS total_price
FROM line_items li
JOIN products p ON p.product_id = li.product_id
GROUP BY li.order_id
ORDER BY li.order_id
LIMIT 5;

SELECT c.customer_name, AVG(t.total_price) AS average_order_price
FROM customers c
JOIN (
	SELECT o.customer_id, li.order_id, SUM(p.price * li.quantity) AS total_price
	FROM orders o
	JOIN line_items li ON li.order_id = o.order_id
	JOIN products p ON p.product_id = li.product_id
	GROUP BY o.customer_id, li.order_id
) AS t ON t.customer_id = c.customer_id
GROUP BY c.customer_id, c.customer_name
ORDER BY c.customer_name;

SELECT e.first_name, e.last_name, COUNT(o.order_id) AS order_count
FROM employees e
JOIN orders o ON o.employee_id = e.employee_id
GROUP BY e.employee_id, e.first_name, e.last_name
HAVING COUNT(o.order_id) > 5
ORDER BY e.last_name;
